# scenes/level_1.py
from scribe.scene import Scene

class Level1(Scene):
    """First level - Forest area"""

    def setup(self):
        # Set background
        self.background = 'forest_bg.png'

        # Add player sprite
        self.player = self.add_sprite(
            'player/idle.png',
            x=100,
            y=400,
            collision=True
        )
        self.player.gravity = True
        self.player.speed = 200

        # Add enemy
        self.enemy = self.add_sprite(
            'enemies/slime.png',
            x=300,
            y=400,
            ai='patrol'
        )

        # Add collectible coin
        self.coin = self.add_sprite(
            'items/coin.png',
            x=200,
            y=350,
            on_collect=self.collect_coin
        )

        # Load tilemap for platforms
        self.tilemap = self.load_tilemap('level1.tmx')

        # Background music
        self.play_music('bgm_forest.mp3', loop=True)

    def update(self, dt):
        # Handle player input
        if self.input.key('right'):
            self.player.x += self.player.speed * dt
            self.player.play_animation('run')
        elif self.input.key('left'):
            self.player.x -= self.player.speed * dt
            self.player.play_animation('run')
        else:
            self.player.play_animation('idle')

        if self.input.key_pressed('space') and self.player.on_ground:
            self.player.jump(500)
            self.play_sound('jump.wav')

        # Check collisions
        if self.player.collides_with(self.enemy):
            self.game_state.health -= 10
            self.camera.shake(intensity=5, duration=0.3)

            if self.game_state.health <= 0:
                self.switch_scene('game_over')

        # Camera follows player
        self.camera.follow(self.player, smooth=True)

    def collect_coin(self, coin):
        """Called when player touches coin"""
        coin.remove()
        self.game_state.score += 10
        self.play_sound('coin.wav')
        self.emit_particles('sparkle', x=coin.x, y=coin.y, count=20)
